function doGet() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var headers = data[0]; var result = []; for (var i = 1; i < data.length; i++) { var row = data[i]; var memberName = row[0]; // A열: 닉네임/이름 if (!memberName) continue; var flowers = []; for (var j = 1; j < headers.length; j++) { var flowerName = headers[j]; var score = Number(row[j]); // 14점 초과(15점 이상) 데이터만 수집 (14점 이하 제외) if (!isNaN(score) && score > 14) { flowers.push({ name: flowerName, score: score }); } } if (flowers.length === 0) continue; // 점수 기준 내림차순 정렬 flowers.sort(function(a, b) { return b.score - a.score; }); // 1. 가장 높은 점수 그룹 (개수 제한 없이 최고점 꽃 전부) var maxScore = flowers[0].score; var top1_flowers = flowers.filter(function(f) { return f.score === maxScore; }); // 2. 두 번째로 높은 점수 그룹 (개수 제한 없이 두 번째 높은 점수 꽃 전부) var remainingFlowers = flowers.filter(function(f) { return f.score < maxScore; }); var top2_flowers = []; if (remainingFlowers.length > 0) { var secondMaxScore = remainingFlowers[0].score; top2_flowers = remainingFlowers.filter(function(f) { return f.score === secondMaxScore; }); } result.push({ name: memberName, top1_flowers: top1_flowers, top2_flowers: top2_flowers }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
